# Firestore Rules – Privacy-first Sharing (Public Links)
#
# This app stores public share links here:
#   artifacts/{APP_ID}/public/data/shares/{token}
#
# Busy-only calendar shares are privacy-safe because they contain ONLY busy time blocks
# (no titles, no locations). External users only need read access to shares/{token}.
#
# IMPORTANT: Adjust rules to your existing ruleset. This is a minimal example.

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function isSignedIn() { return request.auth != null; }

    match /artifacts/{appId}/public/data/shares/{token} {
      // Anyone with the link can read (token is unguessable)
      allow get: if true;
      allow list: if false;

      // Only signed-in users can create/update/revoke links
      allow create, update, delete: if isSignedIn();
    }

    // Optional: calendar-level audit log
    match /artifacts/{appId}/public/data/calendars/{calId}/auditLogs/{logId} {
      allow read: if isSignedIn();
      allow create: if isSignedIn();
      allow update, delete: if false;
    }

    // Optional: personal audit log
    match /artifacts/{appId}/users/{uid}/auditLogs/{logId} {
      allow read, create: if isSignedIn() && request.auth.uid == uid;
      allow update, delete: if false;
    }
  }
}
