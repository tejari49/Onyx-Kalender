// Ergänzung: Event-Kommentare & Polls (Mini-Chat pro Termin)
// Pfade:
// - Private Events: /artifacts/{APP_ID}/users/{uid}/events/{eventId}/comments/*  & /polls/active
// - Geteilte Kalender: /artifacts/{APP_ID}/public/data/calendars/{calId}/events/{eventId}/comments/* & /polls/active
//
// Hinweis: Regeln variieren je nach deinem bestehenden Setup. Diese Snippets sind als Orientierung gedacht.

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function signedIn() { return request.auth != null; }
    function isMe(uid) { return signedIn() && request.auth.uid == uid; }

    // --- Private Events (nur Owner) ---
    match /artifacts/{appId}/users/{uid}/events/{eventId} {
      allow read, write: if isMe(uid);

      match /comments/{commentId} {
        allow read, create: if isMe(uid);
        allow update, delete: if isMe(uid);
      }

      match /polls/{pollId} {
        allow read, write: if isMe(uid);
      }
    }

    // --- Geteilte Kalender ---
    match /artifacts/{appId}/public/data/calendars/{calId} {
      allow read: if signedIn();

      match /events/{eventId} {
        allow read: if signedIn();

        // Kommentare: jeder mit Zugriff kann schreiben (optional: nur write-perm)
        match /comments/{commentId} {
          allow read: if signedIn();
          allow create: if signedIn();
          allow update, delete: if signedIn() && request.auth.uid == resource.data.senderId;
        }

        // Poll: Lesen für alle, Schreiben/Vorschläge idealerweise nur Owner/Write-Perm
        match /polls/{pollId} {
          allow read: if signedIn();
          allow write: if signedIn();
        }
      }
    }
  }
}
